package main
import("go/ast";"go/parser";"go/token";"path/filepath";"strings";"fmt";"os";"sort")
func main(){ fs:=token.NewFileSet(); prod:=map[string]int{}; test:=map[string]int{}; decl:=map[string][]string{}; count:=0
filepath.Walk("backend",func(p string,i os.FileInfo,e error)error{if e!=nil{return e};if i.IsDir()||!strings.HasSuffix(p,".go"){return nil}; f,e:=parser.ParseFile(fs,p,nil,0);if e!=nil{panic(e)};count++; istest:=strings.HasSuffix(p,"_test.go"); ast.Inspect(f,func(n ast.Node)bool{if id,ok:=n.(*ast.Ident);ok{if istest{test[id.Name]++}else{prod[id.Name]++}};return true});if !istest{for _,d:=range f.Decls{if fn,ok:=d.(*ast.FuncDecl);ok{decl[fn.Name.Name]=append(decl[fn.Name.Name],fs.Position(fn.Pos()).String())}}};return nil});names:=[]string{};for n:=range decl{names=append(names,n)};sort.Strings(names);fmt.Println("Go files parsed:",count);for _,n:=range names{if prod[n]==len(decl[n]){fmt.Printf("%s productionRefs=0 testIdentifiers=%d declarations=%v\n",n,test[n],decl[n])}}}
