const ts=require(process.cwd()+'/web/node_modules/typescript'),fs=require('fs'),path=require('path');
let files=[];function walk(p){for(const d of fs.readdirSync(p,{withFileTypes:true})){let q=path.join(p,d.name);if(d.isDirectory())walk(q);else if(/\.tsx?$/.test(q))files.push(q)}}walk('web/src');let prod=new Map(),tests=new Map(),decl=[];
for(const p of files){let s=ts.createSourceFile(p,fs.readFileSync(p,'utf8'),ts.ScriptTarget.Latest,true),test=/\.(test|spec)\.tsx?$/.test(p)||p.includes('/test/');function visit(n){if(ts.isIdentifier(n)){let m=test?tests:prod;m.set(n.text,(m.get(n.text)||0)+1)}if(!test&&ts.isFunctionDeclaration(n)&&n.name)decl.push([n.name.text,p,s.getLineAndCharacterOfPosition(n.getStart()).line+1]);ts.forEachChild(n,visit)}visit(s)}console.log('TS/TSX files parsed:',files.length);for(const [n,p,l]of decl)if(prod.get(n)===1)console.log(`${p}:${l} ${n} productionRefs=0 testIdentifiers=${tests.get(n)||0}`);
